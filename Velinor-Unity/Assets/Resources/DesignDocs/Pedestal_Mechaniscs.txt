Use code with caution.The Player Interaction DetectorAttach this script to your Main Camera (or Player GameObject) to handle raycasting and input.csharpusing UnityEngine;
using TMPro; // Include if using TextMeshPro for UI prompts

public class PlayerInteraction : MonoBehaviour
{
    [SerializeField] private Transform playerCamera;
    [SerializeField] private float rayDistance = 4f;
    [SerializeField] private LayerMask interactableLayer;
    [SerializeField] private TextMeshProUGUI promptText; // UI text element

    private PlayerSystem playerSystem;

    void Start()
    {
        playerSystem = GetComponentInParent<PlayerSystem>();
    }

    void Update()
    {
        // 1. Raycast to look for interactables
        Ray ray = new Ray(playerCamera.position, playerCamera.forward);
        RaycastHit hit;

        if (Physics.Raycast(ray, out hit, rayDistance, interactableLayer))
        {
            Interactable interactable = hit.collider.GetComponent<Interactable>();
            
            if (interactable != null)
            {
                // Display UI Prompt
                if (promptText != null) promptText.text = interactable.promptMessage;

                // 2. Listen for the 'E' key
                if (Input.GetKeyDown(KeyCode.E))
                {
                    interactable.OnInteract(playerSystem);
                }
                return;
            }
        }

        // Clear UI prompt if not looking at an interactable
        if (promptText != null) promptText.text = "";
    }
}

public enum GlyphEmotion { Sorrow, Joy, Anger, Fear }

public class GlyphPedestal : Interactable
{
    [Header("Glyph Properties")]
    [SerializeField] private GlyphEmotion emotionType;
    [SerializeField] private GameObject glyphVisualObject;
    [SerializeField] private Animator pedestalAnimator;
    [SerializeField] private float activationRadius = 5f;

    [Header("Resonance Settings")]
    [SerializeField] private float maxResonanceRadius = 30f;

    private bool isActivated = false;
    private bool isCollected = false;
    private Transform playerTransform;

    void Start()
    {
        promptMessage = $"Press E to absorb {emotionType} Glyph";
        playerTransform = GameObject.FindGameObjectWithTag("Player").transform;
        if(glyphVisualObject != null) glyphVisualObject.SetActive(false); // Dormant initially
    }

    void Update()
    {
        if (isCollected || playerTransform == null) return;

        float distance = Vector3.Distance(transform.position, playerTransform.position);

        // 1. Handle Proximity Activation
        if (distance <= activationRadius && !isActivated)
        {
            ActivatePedestal();
        }

        // 2. Provide data for the Player's Resonance Device (G Key)
        if (distance <= maxResonanceRadius)
        {
            // Normalize proximity strength between 0.0 (edge) and 1.0 (right next to it)
            float resonanceStrength = 1f - (distance / maxResonanceRadius);
            PlayerSystem.Instance.UpdateResonance(resonanceStrength, emotionType);
        }
    }

    private void ActivatePedestal()
    {
        isActivated = true;
        if (glyphVisualObject != null) glyphVisualObject.SetActive(true);
        if (pedestalAnimator != null) pedestalAnimator.SetTrigger("Activate");
    }

    // Overriding the base E-key interaction
    public override void OnInteract(PlayerSystem player)
    {
        if (!isActivated || isCollected) return;

        isCollected = true;
        player.CollectGlyph(emotionType);
        
        // Disable visuals/interactions
        if (glyphVisualObject != null) glyphVisualObject.SetActive(false);
        gameObject.SetActive(false); 
    }
}

using System.Collections.Generic;

public class PlayerSystem : MonoBehaviour
{
    public static PlayerSystem Instance { get; private set; }

    [Header("Movement Specs")]
    [SerializeField] private CharacterController controller;
    [SerializeField] private float baseSpeed = 6f;
    [SerializeField] private float crouchSpeed = 3f;
    [SerializeField] private float crouchHeight = 1f;
    private float originalHeight;
    private float currentSpeed;

    [Header("Resonance Device (G Key)")]
    [SerializeField] private Animator deviceAnimator;
    private bool isDeviceEquipped = false;
    private float activeResonanceStrength = 0f;
    private GlyphEmotion closestGlyphType;

    [Header("Story / RPG Stats")]
    public float empathyStat = 10f;
    public List<GlyphEmotion> collectedGlyphs = new List<GlyphEmotion>();

    void Awake()
    {
        if (Instance == null) Instance = this;
        else Destroy(gameObject);
        
        originalHeight = controller.height;
        currentSpeed = baseSpeed;
    }

    void Update()
    {
        HandleCrouch();
        HandleResonanceDevice();
    }

    // Shift to Crouch (Seamless exploration under crevices)
    private void HandleCrouch()
    {
        if (Input.GetKey(KeyCode.LeftShift))
        {
            controller.height = crouchHeight;
            // Adjust speed if not already slowed down by a Sorrow glyph
            currentSpeed = Mathf.Min(currentSpeed, crouchSpeed);
        }
        else
        {
            controller.height = originalHeight;
            ResetSpeedBasedOnGlyphs();
        }
    }

    // G Key Animation Logic
    private void HandleResonanceDevice()
    {
        if (Input.GetKeyDown(KeyCode.G))
        {
            isDeviceEquipped = !isDeviceEquipped;
            deviceAnimator.SetBool("IsInspecting", isDeviceEquipped);
        }

        if (isDeviceEquipped && activeResonanceStrength > 0.05f)
        {
            // PRO-TIP: Use activeResonanceStrength here to drive a UI meter fill amount, 
            // post-processing screen flashes, or audio ping pitch frequencies.
            Debug.Log($"Device pulsing at {activeResonanceStrength * 100}% for {closestGlyphType} Glyph!");
        }
        
        // Decay resonance reading if moving away
        activeResonanceStrength = Mathf.Lerp(activeResonanceStrength, 0f, Time.deltaTime);
    }

    public void UpdateResonance(float strength, GlyphEmotion type)
    {
        if (strength > activeResonanceStrength)
        {
            activeResonanceStrength = strength;
            closestGlyphType = type;
        }
    }

    // Handle the mechanical feedback of absorbing Glyphs
    public void CollectGlyph(GlyphEmotion emotion)
    {
        collectedGlyphs.Add(emotion);
        Debug.Log($"Absorbed {emotion} Glyph!");

        if (emotion == GlyphEmotion.Sorrow)
        {
            baseSpeed *= 0.7f; // Permanently slow down base movement
            empathyStat += 25f; // Drastically improve listening capabilities
            ResetSpeedBasedOnGlyphs();
        }
        // Add other emotional stat impacts here...
    }

    private void ResetSpeedBasedOnGlyphs()
    {
        currentSpeed = Input.GetKey(KeyCode.LeftShift) ? crouchSpeed : baseSpeed;
    }
}
